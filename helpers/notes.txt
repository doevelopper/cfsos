# Function to check file permissions
check_file_permissions = \
        if [ ! -f "$1" ]; then \
                echo "ERROR: File '$1' does not exist."; \
                exit 1; \
        elif [ "$(stat -c %a "$1")" != "600" ]; then \
                echo "ERROR: File '$1' has incorrect permissions. Must be 600 (currently $(stat -c %a "$1"))."; \
                exit 1; \
        fi

$(foreach defconfig,$(SUPPORTED_TARGETS),$(defconfig)-rauc-certs): %-rauc-certs: $(BASE)/root/ca.cert.pem $(BASE)/meta/%.meta
	$(Q)$(call MESSAGE,"[ Create $*'s RAUC signing key and certificate ]")
	@. $(BASE)/meta/$*.meta
	@cd $(RAUC_DIR) && openssl req -newkey -config $(OPENSSL_CONF) rsa:4096 \
			-keyout $(BASE)/private/$*.key.pem \
			-out $(BASE)/$*.csr.pem \
			-subj "/O=$(TARGET_ORG)/CN=$(TARGET_CN)" > /dev/null
	@echo "[OK] Private key '$(BASE)/private/$*.key.pem' created."
	@cd $(RAUC_DIR) && openssl ca -batch -config $(OPENSSL_CONF) -extensions v3_leaf \
			-in $(BASE)/$*.csr.pem \
			-out $(BASE)/$*.cert.pem \
			-keyfile $(BASE)/root/private/ca.key.pem > /dev/null
	@echo "[OK] Certificate '$(BASE)/$*.cert.pem' created."

	@rm -rvf $(BASE)/$* $(BASE)/private/$*.key.pem
